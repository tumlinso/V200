cmake_minimum_required(VERSION 3.18)
project(v200_bootstrap_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(v200_planning_test tests/planning_test.cpp)
target_include_directories(v200_planning_test PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(v200_planning_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME cpu_planning COMMAND v200_planning_test)
option(V200_BUILD_CUDA_PROBE "Build (do not run) the explicit Driver API pair probe" OFF)
if(V200_BUILD_CUDA_PROBE)
  find_package(CUDAToolkit 12 REQUIRED)
  if(CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.0")
    message(FATAL_ERROR "Qualify with CUDA12.x for this Volta example; no silent toolkit substitution")
  endif()
  add_executable(v200_pair_probe cuda/pair_vmm_probe.cpp)
  target_include_directories(v200_pair_probe PRIVATE include)
  target_link_libraries(v200_pair_probe PRIVATE CUDA::cuda_driver)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(v200_pair_probe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  # Intentionally no GPU ctest: execution belongs to an admitted bundle.
endif()
